const express = require('express'); const fs = require('fs'); const pdf = require('pdf-parse'); const app = express(); app.use(express.urlencoded({ extended: true })); let resultData = {}; // এখানে সব রোল আর রেজাল্ট জমা থাকবে // ১. PDF ফাইলটি পড়ে ডেটা অবজেক্টে সেভ করার ফাংশন const loadPDFData = async () => { let dataBuffer = fs.readFileSync('result.pdf'); // আপনার PDF ফাইলের নাম const data = await pdf(dataBuffer); // PDF-এর টেক্সটকে লাইন অনুযায়ী ভাগ করা const lines = data.text.split('\n'); lines.forEach(line => { // ধরুন লাইনে রোল ৬ সংখ্যার এবং তারপর রেজাল্ট আছে // Regex ব্যবহার করে রোল এবং জিপিএ খুঁজে বের করা (এটি PDF এর ফরম্যাটের ওপর নির্ভর করবে) const match = line.match(/(\d{6})\s+([\d.]+)/); if (match) { resultData[match[1]] = match[2]; // match[1] = Roll, match[2] = Result } }); console.log("PDF Data Loaded!"); }; // ২. হোম পেজ (HTML) app.get('/', (req, res) => { res.send(`

Diploma Result Search

`); }); // ৩. সার্চ রেজাল্ট পেজ app.post('/search', (req, res) => { const roll = req.body.roll; const result = resultData[roll]; if (result) { res.send(`

Roll: ${roll}

Result: ${result}

Back`); } else { res.send(`

Result Not Found!

Back`); } }); // সার্ভার চালু করা loadPDFData().then(() => { app.listen(3000, () => console.log('Server running at http://localhost:3000')); });
